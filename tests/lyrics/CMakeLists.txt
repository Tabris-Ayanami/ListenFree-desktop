cmake_minimum_required(VERSION 3.28)
project(ListenFreeLyricsChecks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.11 REQUIRED COMPONENTS QuickTest Quick Qml ShaderTools GuiPrivate)
set(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../..")
set(LYRIC_NATIVE_DIR "${REPO}/src/qmlbridge" CACHE PATH "Native headers for matched baseline replay")
set(LYRIC_SHADER_DIR "${REPO}/music_player_desktop/shaders" CACHE PATH "Shaders for matched baseline replay")
qt_add_executable(listenfree_lyrics_tests runner.cpp)
qt_add_executable(listenfree_lyrics_render render_probe.cpp)
target_link_libraries(listenfree_lyrics_tests PRIVATE Qt6::QuickTest)
target_link_libraries(listenfree_lyrics_render PRIVATE Qt6::GuiPrivate)
foreach(target listenfree_lyrics_tests listenfree_lyrics_render)
    target_sources(${target} PRIVATE "${LYRIC_NATIVE_DIR}/spring_value.h" "${LYRIC_NATIVE_DIR}/lyric_text_metrics.h")
    target_include_directories(${target} PRIVATE "${LYRIC_NATIVE_DIR}")
    target_link_libraries(${target} PRIVATE Qt6::Quick Qt6::Qml)
    if(WIN32)
        target_link_libraries(${target} PRIVATE psapi)
    endif()
    qt_add_shaders(${target} ${target}_shaders PREFIX "/shaders"
        BASE "${LYRIC_SHADER_DIR}"
        FILES "${LYRIC_SHADER_DIR}/lyric-word.frag"
              "${LYRIC_SHADER_DIR}/lyric-grapheme.frag")
endforeach()
